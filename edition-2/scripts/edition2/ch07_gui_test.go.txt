// Independent optional-module projection probes from Chapter 7's literal fixtures.
package gui_test

import (
	"encoding/json"
	ens "example.com/ensemble"
	gui "example.com/ensemble-gui"
	"io"
	"reflect"
	"testing"
)

func c7Owner(t *testing.T) *gui.Server {
	t.Helper()
	e := ens.New(io.Discard)
	t.Cleanup(func() { _ = e.Close() })
	s, err := gui.NewServer(e, "fixture-agent", "http://127.0.0.1:8088", nil)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = s.Close() })
	return s
}
func c7Decode(t *testing.T, s string) any {
	t.Helper()
	var out any
	if err := json.Unmarshal([]byte(s), &out); err != nil {
		t.Fatal(err)
	}
	return out
}
func c7AssertJSON(t *testing.T, got any, want string) {
	t.Helper()
	raw, err := json.Marshal(got)
	if err != nil {
		t.Fatal(err)
	}
	if !reflect.DeepEqual(c7Decode(t, string(raw)), c7Decode(t, want)) {
		t.Fatalf("typed display projection changed meaning: got=%s want=%s", raw, want)
	}
}
func TestC7GUIProjection(t *testing.T) {
	owner := c7Owner(t)
	const text = `{"type":"text","text":"","from":{"vendor":"gemini","model":"fixture-gemini","surface":"generate_content"},"opaque":"text-signature"}`
	const opaque = `{"type":"opaque","from":{"vendor":"gemini","model":"fixture-gemini","surface":"generate_content"},"data":{"nested":{"thoughtSignature":"hidden-bytes"}}}`
	const safe = `{"type":"text","text":"","from":{"vendor":"gemini","model":"fixture-gemini","surface":"generate_content"}}`
	var p ens.Part
	if err := json.Unmarshal([]byte(text), &p); err != nil {
		t.Fatal(err)
	}
	before, _ := json.Marshal(p)
	c7AssertJSON(t, gui.ProjectPart(owner, p), safe)
	after, _ := json.Marshal(p)
	if string(before) != string(after) {
		t.Fatal("display projection mutated provider replay part")
	}
	if err := json.Unmarshal([]byte(opaque), &p); err != nil {
		t.Fatal(err)
	}
	c7AssertJSON(t, gui.ProjectPart(owner, p), `{"type":"opaque","placeholder":true}`)
	var e ens.Event
	if err := json.Unmarshal([]byte(`{"seq":8,"type":"response_ended","response":{"parts":[`+text+`,`+opaque+`]}}`), &e); err != nil {
		t.Fatal(err)
	}
	response := gui.ProjectEvent(owner, e)["response"].(map[string]any)
	c7AssertJSON(t, response["parts"], `[`+safe+`,{"type":"opaque","placeholder":true}]`)
	if err := json.Unmarshal([]byte(`{"seq":9,"type":"message_received","message":{"actor":"user","parts":[`+text+`]}}`), &e); err != nil {
		t.Fatal(err)
	}
	c7AssertJSON(t, gui.ProjectEvent(owner, e)["message"].(map[string]any)["parts"], `[`+safe+`]`)
	t.Run("tool-result-explicit-false", func(t *testing.T) {
		var result ens.Event
		if err := json.Unmarshal([]byte(`{"seq":10,"type":"tool_returned","tool":{"call_id":"read-a","parts":[`+text+`],"is_error":false}}`), &result); err != nil {
			t.Fatal(err)
		}
		c7AssertJSON(t, gui.ProjectEvent(owner, result)["tool"], `{"call_id":"read-a","parts":[`+safe+`],"is_error":false}`)
	})
	t.Run("nested-result", func(t *testing.T) {
		var result ens.Part
		if err := json.Unmarshal([]byte(`{"type":"tool_result","call_id":"read-a","parts":[`+text+`],"is_error":true}`), &result); err != nil {
			t.Fatal(err)
		}
		c7AssertJSON(t, gui.ProjectPart(owner, result), `{"type":"tool_result","call_id":"read-a","parts":[`+safe+`],"is_error":true}`)
	})
	t.Run("argument-names-are-data", func(t *testing.T) {
		const input = `{"type":"tool_call","call_id":"data-a","name":"read_file","args":{"opaque":"keep-as-argument","thoughtSignature":"also-user-data","nested":{"opaque":"still-data"}},"opaque":"bound-call-signature"}`
		var call ens.Part
		if err := json.Unmarshal([]byte(input), &call); err != nil {
			t.Fatal(err)
		}
		c7AssertJSON(t, gui.ProjectPart(owner, call), `{"type":"tool_call","call_id":"data-a","name":"read_file","args":{"opaque":"keep-as-argument","thoughtSignature":"also-user-data","nested":{"opaque":"still-data"}}}`)
	})
}
