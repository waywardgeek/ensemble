package ensemble

import (
	"net/http"
	"reflect"
	"testing"
	"time"
)

type reviewRetainedObserver struct{ client []byte }

func (*reviewRetainedObserver) Observe(Observation) {}

func TestReviewSubscribeRejectsClosedOwner(t *testing.T) {
	app, a := actorAgent(t, func(w http.ResponseWriter, r *http.Request) { replyText(w, "unused") })
	if err := app.Close(); err != nil {
		t.Fatal(err)
	}
	id, err := app.Subscribe(a.ID(), &reviewRetainedObserver{})
	if err == nil {
		app.Unsubscribe(id)
		t.Fatal("closed Ensemble admitted a new subscription worker")
	}
}

// Inspect retained values, rather than requiring an identifier or particular
// tombstone design. The mutex/map access is an explicit fixture adapter.
func reviewDeliveryReferences(v reflect.Value, seen map[uintptr]bool) bool {
	if !v.IsValid() {
		return false
	}
	if v.Type() == reflect.TypeOf((*reviewRetainedObserver)(nil)) {
		return !v.IsNil()
	}
	switch v.Kind() {
	case reflect.Interface:
		if !v.IsNil() {
			return reviewDeliveryReferences(v.Elem(), seen)
		}
	case reflect.Pointer:
		if v.IsNil() || seen[v.Pointer()] {
			return false
		}
		seen[v.Pointer()] = true
		return reviewDeliveryReferences(v.Elem(), seen)
	case reflect.Chan:
		return v.Type().Elem() == reflect.TypeOf(Observation{}) && !v.IsNil()
	case reflect.Struct:
		for i := 0; i < v.NumField(); i++ {
			if reviewDeliveryReferences(v.Field(i), seen) {
				return true
			}
		}
	case reflect.Map:
		it := v.MapRange()
		for it.Next() {
			if reviewDeliveryReferences(it.Value(), seen) {
				return true
			}
		}
	}
	return false
}
func TestReviewSubscriptionReleasesDeliveryReferences(t *testing.T) {
	app, a := actorAgent(t, func(w http.ResponseWriter, r *http.Request) { replyText(w, "unused") })
	id, err := app.Subscribe(a.ID(), &reviewRetainedObserver{client: make([]byte, 1<<20)})
	if err != nil {
		t.Fatal(err)
	}
	app.Unsubscribe(id)
	deadline := time.Now().Add(time.Second)
	for {
		app.mu.Lock()
		// Mark the parent as visited: this check concerns this owner's closed
		// subscription records, not cycles through unrelated live Agents.
		retained := reviewDeliveryReferences(reflect.ValueOf(app.observers), map[uintptr]bool{reflect.ValueOf(app).Pointer(): true})
		app.mu.Unlock()
		if !retained {
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("drained subscription retains observer/client or delivery queue")
		}
		time.Sleep(time.Millisecond)
	}
	if reason := app.SubscriptionStatus(id); reason == "" || reason == "unknown subscription" {
		t.Fatalf("closure reason lost: %q", reason)
	}
}
