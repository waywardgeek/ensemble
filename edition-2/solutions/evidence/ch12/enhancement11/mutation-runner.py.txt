import pathlib,subprocess,tempfile,shutil,os,json,difflib,time
root=pathlib.Path.cwd();source=root/'solutions/edition-2/ch11';out=root/'solutions/edition-2/evidence/ch12/enhancement11'
scratch=pathlib.Path(tempfile.mkdtemp(prefix='ensemble-ch11-mutant-'));shutil.copytree(source,scratch,dirs_exist_ok=True)
client='internal/mcp/client.go';manager='internal/mcp/mcp.go';agent='internal/agent/agent.go';auto='internal/agent/mcp.go'
cases=[
 ('initialized-notification',client,'Method: "notifications/initialized"','Method: "notifications/not-initialized"','mcp-handshake'),
 ('reverse-id',client,'response := message{ID: m.ID}','response := message{ID: raw(1)}','reverse-call'),
 ('reverse-empty',client,'response.Result = raw(toolResult{Content:', 'if params.Name == "read_file" { text = "" }; response.Result = raw(toolResult{Content:','reverse-call'),
 ('reverse-error-flag',client,'IsError: err != nil','IsError: false','reverse-call'),
 ('hidden-dispatch',agent,'(automatic || tool.Ephemeral == "")','true','ephemeral-round'),
 ('missing-discovery',agent,'range a.mcp.Tools()','range []common.ToolDefinition{}','tool-discovery'),
 ('visible-automatic',agent,'tool.Ephemeral != "" || !a.toolEnabled(tool.Name)','!a.toolEnabled(tool.Name)','tool-discovery'),
 ('description',manager,'Description: info.Description','Description: ""','tool-discovery'),
 ('schema',manager,'Schema: info.Schema','Schema: json.RawMessage(`{}`)','tool-discovery'),
 ('arguments',client,'"arguments": args','"arguments": raw(map[string]any{})','mcp-tool-call'),
 ('empty-result',client,'strings.Join(text, "\\n")','strings.Join(text[:0], "\\n")','mcp-tool-call'),
 ('second-result-block',client,'strings.Join(text, "\\n")','strings.Join(text[:min(1,len(text))], "\\n")','mcp-tool-call'),
 ('tool-error-flag',client,'if result.IsError {','if false && result.IsError {','mcp-tool-call'),
 ('missing-round',auto,'tool.Ephemeral == "round"','false','ephemeral-round'),
 ('missing-turn',auto,'tool.Ephemeral == "turn"','false','ephemeral-round'),
 ('turn-every-round',auto,' && !t.ephemeralTurn','','ephemeral-round'),
 ('stale-ephemera','internal/history/reduce.go','c.Ephemera = nil','c.Ephemera = c.Ephemera','ephemeral-round'),
 ('failed-auto-context',auto,'r.owner.Ensemble().Logf("ephemeral tool %s: %v", name, err)','r.owner.Ensemble().Logf("ephemeral tool %s: %v", name, err); parts = append(parts, err.Error())','ephemeral-round'),
 ('no-stdio',manager,'t, err = startStdio(m.parent, server)','return nil','tool-discovery'),
 ('ws-outgoing','gui/mcp.go','"payload": message','"data": message','ws-tunnel'),
 ('ws-incoming','gui/gui.go','case tunnel.incoming <- input.Payload:','case tunnel.incoming <- json.RawMessage(`null`):','ws-tunnel'),
]
results=[]
def run(label):
 env=dict(os.environ,MCP_PROBE_CORE=str(scratch),MCP_PROBE_GUI=str(scratch/'gui'))
 started=time.monotonic();p=subprocess.run(['/tmp/ensemble-ch11-probe','-test.run','^TestChapter11Probe$','-test.v'],env=env,text=True,capture_output=True,timeout=60)
 (out/(label+'.txt')).write_text(p.stdout+p.stderr)
 records=[json.loads(line) for line in p.stdout.splitlines() if line.startswith('[{')]
 return p.returncode,records[0] if records else None,round(time.monotonic()-started,3)
try:
 code,checks,elapsed=run('mutation-baseline');assert code==0 and all(c['passed'] for c in checks)
 for name,path,old,new,intended in cases:
  file=scratch/path;before=file.read_text();assert before.count(old)==1,(name,before.count(old),old)
  after=before.replace(old,new);file.write_text(after)
  (out/('mutation-'+name+'.patch')).write_text(''.join(difflib.unified_diff(before.splitlines(True),after.splitlines(True),fromfile=path,tofile=path)))
  code,checks,elapsed=run('mutation-'+name)
  failures=[c for c in checks or [] if not c['passed']]
  killed=code==0 and any(c['id']==intended for c in failures)
  results.append(dict(name=name,intended=intended,killed=killed,process_exit=code,seconds=elapsed,failures=failures))
  (out/'mutations.json').write_text(json.dumps(results,indent=2)+'\n')
  print(name,'KILLED' if killed else 'INVALID/SURVIVED',flush=True)
  file.write_text(before)
 code,checks,elapsed=run('mutation-restored');assert code==0 and all(c['passed'] for c in checks)
finally:shutil.rmtree(scratch)
