func TestInvoiceDuplicatePair(t *testing.T){
 raw,err:=os.ReadFile(os.Getenv("GRADE17_WIRE"));if err!=nil{t.Fatal(err)}
 chunks:=strings.Split(string(raw),"===== REQUEST ")[1:]
 var reqs []fakevendor.Recorded
 for _,s:=range chunks[:2]{body:=s[strings.Index(s,"\n")+1:];reqs=append(reqs,fakevendor.Recorded{Body:[]byte(body)})}
 dir:=t.TempDir();both:=append(append([]byte(nil),reqs[0].Body...),reqs[1].Body...)
 os.WriteFile(filepath.Join(dir,"current"),both,0644);os.WriteFile(filepath.Join(dir,"prior"),both,0644)
 r:=contextResult{Errs:map[string]string{}}
 invoiceLens(&r,invoiceOut{reqs:reqs},dir,"current","prior")
 if r.Errs["cache-lens-exists"]=="" {t.Fatal("identical accumulated files passed the lens rotation check")}
}
