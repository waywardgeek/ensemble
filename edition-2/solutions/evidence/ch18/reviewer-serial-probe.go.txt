package ensemble_test
import("context";"fmt";"net/http";"net/http/httptest";"testing";"time";"ensemble/ensemble")
func TestReviewerCacheSerial(t *testing.T){
 for n:=0;n<12;n++ {
 w:= &cacheGateWriter{entered:make(chan struct{}),release:make(chan struct{})}
 s:=httptest.NewServer(http.HandlerFunc(func(rw http.ResponseWriter,r *http.Request){fmt.Fprint(rw,`{"content":[{"type":"text","text":"ok"}],"usage":{"input_tokens":10,"output_tokens":1}}`)}))
 a,e:=ensemble.New(w).NewAgent(ensemble.Config{Model:"claude-haiku-5-5",BaseURL:s.URL,DataDir:t.TempDir(),DisableStreaming:true});if e!=nil{t.Fatal(e)}
 ctx,cancel:=context.WithTimeout(context.Background(),2*time.Second)
 first:=make(chan error,1);go func(){_,e:=a.Ask(ctx,"first");first<-e}()
 select{case <-w.entered:case <-ctx.Done():t.Fatal("first request never reached lens")}
 close(w.release);select{case e:=<-first:if e!=nil{t.Fatal(e)};case <-ctx.Done():t.Fatal("first blocked")}
 second:=make(chan error,1);go func(){_,e:=a.Ask(ctx,"second");second<-e}()
 
 select{case e:=<-second:if e!=nil{t.Fatal(e)};case <-ctx.Done():t.Fatal("new request blocked")}
 a.Shutdown();cancel();s.Close()
 }
}
