package gui

import (
 "encoding/json"
 "ensemble/ensemble"
 "fmt"
 "io"
 "net"
 "net/http/httptest"
 "strings"
 "testing"
 "time"
 "github.com/gorilla/websocket"
)

func TestReviewerRelayDisconnects(t *testing.T) {
 root:=ensemble.New(io.Discard)
 a,err:=root.NewAgent(ensemble.Config{Vendor:ensemble.OpenAI,Model:"gpt-4o-mini",DataDir:t.TempDir()})
 if err!=nil {t.Fatal(err)}
 defer a.Shutdown()
 hub,err:=New(root,a,io.Discard);if err!=nil {t.Fatal(err)}
 defer hub.Close()
 server:=httptest.NewServer(hub.Handler(""));defer server.Close()
 ws,_,err:=websocket.DefaultDialer.Dial("ws"+strings.TrimPrefix(server.URL,"http")+"/ws",nil)
 if err!=nil {t.Fatal(err)}
 defer ws.Close();ws.SetReadDeadline(time.Now().Add(4*time.Second))
 var tunnel *WSTransport
 deadline:=time.Now().Add(time.Second)
 for tunnel==nil && time.Now().Before(deadline) {
  hub.mu.Lock();for _,x:=range hub.tunnels {tunnel=x};hub.mu.Unlock()
  if tunnel==nil {time.Sleep(time.Millisecond)}
 }
 if tunnel==nil {t.Fatal("browser never registered")}
 listener,err:=net.Listen("tcp","127.0.0.1:0");if err!=nil {t.Fatal(err)}
 defer listener.Close();go hub.ServeMCP(listener)
 dial:=func() net.Conn {c,e:=net.Dial("tcp",listener.Addr().String());if e!=nil {t.Fatal(e)};c.SetDeadline(time.Now().Add(4*time.Second));t.Cleanup(func(){c.Close()});return c}
 readRequest:=func() map[string]any {var frame struct {Payload map[string]any `json:"payload"`};if e:=ws.ReadJSON(&frame);e!=nil {t.Fatal(e)};if frame.Payload["id"]==nil {t.Fatal("relay omitted correlation ID")};return frame.Payload}
 first:=dial();fmt.Fprintln(first,`{"jsonrpc":"2.0","id":0,"method":"tools/list"}`)
 old:=readRequest();first.Close()
 // Actual client EOF must cancel and remove the parked request before a late reply.
 deadline=time.Now().Add(time.Second);count:=1
 for count!=0 && time.Now().Before(deadline) {tunnel.relayMu.Lock();count=len(tunnel.pending);tunnel.relayMu.Unlock();if count!=0 {time.Sleep(time.Millisecond)}}
 if count!=0 {t.Fatal("client EOF retained parked relay request")}
 if err:=ws.WriteJSON(map[string]any{"type":"jsonrpc","payload":map[string]any{"jsonrpc":"2.0","id":old["id"],"result":map[string]any{"late":true}}});err!=nil {t.Fatal(err)}
 second:=dial();fmt.Fprintln(second,`{"jsonrpc":"2.0","id":"still","method":"tools/list"}`)
 readRequest() // A late cancelled reply must not block the browser reader.
 ws.Close()
 var response map[string]any
 if err:=json.NewDecoder(second).Decode(&response);err!=nil {t.Fatal(err)}
 if response["id"]!="still" || response["error"]==nil {t.Fatalf("browser disconnect not correlated: %v",response)}
 t.Log("client EOF cleaned pending request; late reply did not block; browser EOF returned correlated error")
}
